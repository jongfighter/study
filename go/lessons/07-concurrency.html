    핵심 개념 한국어 요약

    동시성(concurrency)과 병렬성(parallelism). 동시성은 일을 "여러 조각으로 나누어 구조를 잡는 것"이고, 병렬성은 그 조각을 "실제로 동시에 실행하는 것"입니다(100 Go Mistakes #55의 정리). Go는 동시성을 언어 기능으로 제공하고, 몇 개를 실제로 병렬 실행할지는 런타임이 정합니다.

    고루틴(goroutine). 함수 호출 앞에 go를 붙이면 그 호출이 별도 실행 흐름으로 시작됩니다. 호출한 쪽은 기다리지 않고 바로 다음 줄로 갑니다. 스레드보다 훨씬 가벼워서 수천 개를 만드는 것이 흔합니다. 단, main 함수가 끝나면 아직 실행 중인 고루틴이 있어도 프로그램이 끝납니다. 그래서 끝나기를 기다리는 수단이 필요합니다.

    채널(channel). 고루틴 사이에서 값을 주고받는 통로입니다. ch <- v는 보내기, v := <-ch는 받기입니다. 버퍼 없는 채널은 보내는 쪽과 받는 쪽이 만나야 둘 다 진행하므로 동기화 수단이 되고, 버퍼 채널은 정해진 개수까지는 받는 쪽이 없어도 보낼 수 있습니다. 보내는 쪽이 close(ch)로 닫으면 받는 쪽의 for v := range ch가 끝납니다.

    select. 여러 채널 연산 중 준비된 것을 골라 실행하는 문장입니다. 준비된 것이 여럿이면 무작위로 하나를 고르므로 순서에 의존하면 안 됩니다(100 Go Mistakes #64). time.After를 한 갈래로 넣으면 타임아웃이 됩니다.

    데이터 레이스(data race). 여러 고루틴이 같은 메모리를 동시에 읽고 쓰는데 그중 하나 이상이 쓰기이면 결과를 믿을 수 없습니다. 해법은 sync.Mutex로 한 번에 한 고루틴만 접근하게 하거나, 숫자 하나라면 sync/atomic을 쓰는 것입니다. 맵은 동시에 쓰면 Go 런타임이 fatal error로 프로그램을 멈춰 줄 때가 있지만, 그것을 믿고 안전하다고 생각하면 안 됩니다. 채널이냐 뮤텍스냐는 "소유권을 넘기는 협업이면 채널, 같은 상태를 여럿이 건드리면 뮤텍스"가 출발점입니다(Learn Go with Tests의 sync 장 요약, 100 Go Mistakes #57).

    context. 취소 신호, 마감 시각(deadline), 요청 범위 값을 호출 사슬을 따라 전달하는 표준 방법입니다. ctx.Done()은 취소되거나 마감이 지나면 닫히는 채널이고, ctx.Err()는 이유(context canceled, context deadline exceeded)를 알려 줍니다. 시간이 걸리는 함수는 첫 번째 인자로 ctx context.Context를 받는 것이 관례입니다. 취소는 "협조적"이라서 받은 쪽이 ctx.Done()을 확인해야 멈춥니다.

    이 레슨의 Go 버전 메모이 코스의 lab/go.mod는 go 1.24입니다. 반복문 변수는 매 반복마다 새로 만들어지므로(Go 1.22부터) for 안의 go func() { ... }()가 변수를 그대로 써도 안전합니다. 100 Go Mistakes #63이 "이제 해당 없음"이라고 한 문제입니다.

    2. WaitGroup으로 모두 끝날 때까지 기다리기

      1단계에서 만든 mine/main.go의 내용을 아래 코드로 바꿉니다. wg.Add(1)은 고루틴을 시작하기 전에 호출하고, 고루틴 안에서는 defer wg.Done()으로 끝났음을 알립니다. WaitGroup은 복사하면 안 되므로 포인터(&wg)로 넘깁니다.

      package main

import (
	"fmt"
	"sync"
)

func worker(id int, wg *sync.WaitGroup) {
	defer wg.Done()
	fmt.Println("작업", id, "시작")
	fmt.Println("작업", id, "끝")
}

func main() {
	var wg sync.WaitGroup
	for id := 1; id <= 3; id++ {
		wg.Add(1)
		go worker(id, &wg)
	}
	wg.Wait()
	fmt.Println("모든 작업 완료")
}

    3. 채널로 값 주고받기: 파이프라인과 버퍼 채널

      mine/main.go를 아래 코드로 바꿉니다. generate는 1부터 n까지 보내고, square는 받은 값을 제곱해 다음 채널로 보냅니다. 함수가 돌려주는 <-chan int는 "받기 전용 채널"이라 이 함수 밖에서는 보낼 수 없습니다. 보내는 쪽이 defer close(out)으로 닫아 주므로 for v := range가 끝납니다.

      package main

import "fmt"

func generate(n int) <-chan int {
	out := make(chan int)
	go func() {
		defer close(out)
		for i := 1; i <= n; i++ {
			out <- i
		}
	}()
	return out
}

func square(in <-chan int) <-chan int {
	out := make(chan int)
	go func() {
		defer close(out)
		for v := range in {
			out <- v * v
		}
	}()
	return out
}

func main() {
	for v := range square(generate(5)) {
		fmt.Println("제곱:", v)
	}

	buffered := make(chan string, 2)
	buffered <- "첫째"
	buffered <- "둘째"
	fmt.Println("버퍼 채널 len/cap:", len(buffered), cap(buffered))
	fmt.Println(<-buffered, <-buffered)
}

      이 출력은 보내는 쪽과 받는 쪽이 하나씩이라 순서가 고정되므로 예상 출력으로 믿어도 됩니다. 마지막 두 줄은 버퍼 크기 2인 채널입니다. 받는 쪽 없이도 2개까지 보낼 수 있었고(len은 지금 들어 있는 개수, cap은 용량), 세 번째를 보내려 하면 누군가 받을 때까지 멈춥니다.

    4. select와 time.After로 타임아웃 만들기

      mine/main.go를 아래 코드로 바꿉니다. slowWork는 200ms 뒤에 결과를 보내는 작업입니다. 결과 채널을 버퍼 1로 만든 이유는, 타임아웃으로 아무도 받지 않아도 작업 고루틴이 보내다 멈춰서 영원히 남는 일(고루틴 누수)이 없게 하려는 것입니다.

      package main

import (
	"fmt"
	"time"
)

func slowWork() <-chan string {
	result := make(chan string, 1)
	go func() {
		time.Sleep(200 * time.Millisecond)
		result <- "작업 결과"
	}()
	return result
}

func waitFor(timeout time.Duration) {
	select {
	case r := <-slowWork():
		fmt.Println("성공:", r)
	case <-time.After(timeout):
		fmt.Println("시간 초과:", timeout)
	}
}

func main() {
	waitFor(50 * time.Millisecond)
	waitFor(500 * time.Millisecond)
}

성공: 작업 결과

      50ms는 작업이 끝나기 전에 시간이 다 되어 시간 초과, 500ms는 작업이 먼저 끝나 성공입니다. 시간 차이를 넉넉하게 둔 덕분에 이 출력은 거의 항상 같습니다. 타이머로 결과를 고정할 때는 이렇게 차이를 크게 두는 것이 안전합니다.

    5. 공유 변수를 보호하지 않으면: 잠금 없음, Mutex, atomic 비교

      mine/main.go를 아래 코드로 바꿉니다. 고루틴 100개가 각각 1만 번 카운터를 올립니다. 기대값은 100 × 10000 = 1000000입니다. 세 방식을 같은 틀(count 함수)에 끼워 비교합니다. 첫 번째는 일부러 잘못된 코드입니다.

      package main

import (
	"fmt"
	"sync"
	"sync/atomic"
)

const (
	goroutines = 100
	perWorker  = 10000
)

func count(inc func()) {
	var wg sync.WaitGroup
	for range goroutines {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for range perWorker {
				inc()
			}
		}()
	}
	wg.Wait()
}

func main() {
	want := goroutines * perWorker

	unsafe := 0
	count(func() { unsafe++ })
	fmt.Printf("잠금 없음: %d (기대값 %d)\n", unsafe, want)

	var mu sync.Mutex
	locked := 0
	count(func() {
		mu.Lock()
		locked++
		mu.Unlock()
	})
	fmt.Printf("Mutex: %d\n", locked)

	var total atomic.Int64
	count(func() { total.Add(1) })
	fmt.Printf("atomic: %d\n", total.Load())
}

      func Map(ctx context.Context, workers int, nums []int, fn Func) ([]int, error) {
	if workers < 1 {
		workers = 1
	}
	runCtx, cancel := context.WithCancel(ctx)
	defer cancel()

	results := make([]int, len(nums))
	jobs := make(chan int)

	var (
		wg       sync.WaitGroup
		mu       sync.Mutex
		firstErr error
	)
	for range workers {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for i := range jobs {
				v, err := fn(runCtx, nums[i])
				if err != nil {
					mu.Lock()
					if firstErr == nil {
						firstErr = err
						cancel()
					}
					mu.Unlock()
					continue
				}
				results[i] = v
			}
		}()
	}

feed:
	for i := range nums {
		select {
		case jobs <- i:
		case <-runCtx.Done():
			break feed
		}
	}
	close(jobs)
	wg.Wait()

	if firstErr != nil {
		return nil, firstErr
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	return results, nil
}

      break feed의 feed:는 라벨(label)입니다. select 안에서 그냥 break를 쓰면 for가 아니라 select만 빠져나오기 때문에, 바깥 반복문을 끝내려고 라벨을 붙였습니다(100 Go Mistakes #34).

    익숙함이 만드는 함정 ①: 구조적 동시성이 없습니다Kotlin의 coroutineScope는 자식 코루틴이 모두 끝나야 빠져나오고, 부모가 취소되면 자식도 취소됩니다. Go의 고루틴에는 부모와 자식 관계가 없습니다. go로 시작한 고루틴은 아무도 기다리지 않으면 main이 끝날 때 말없이 사라지고(1단계), 아무도 멈추지 않으면 계속 남습니다(고루틴 누수). "고루틴을 시작할 때는 언제 끝날지 계획이 있어야 한다"가 규칙입니다(100 Go Mistakes #62).

func TestParallelSum(t *testing.T) {
	thousand := make([]int, 1000)
	for i := range thousand {
		thousand[i] = i + 1
	}

	tests := []struct {

		nums  []int
		parts int
		want  int
	}{

    힌트 보기

    조각 크기는 (len(nums) + parts - 1) / parts로 올림 계산하고, nums[start:end]로 자릅니다. 고루틴을 몇 개 시작했는지 세어 두었다가 같은 횟수만큼 채널에서 받습니다. parts가 0 이하이면 1로, 원소 수보다 크면 원소 수로 보정하세요. 고루틴에는 조각을 인자로 넘겨 주면 어떤 반복에서 만든 조각인지 헷갈리지 않습니다.

    정답 예시 보기 
    // Package exercises는 레슨 07 연습 문제의 정답 예시입니다.
package exercises

// ParallelSum은 nums를 parts개 조각으로 나눠 고루틴마다 합을 구한 뒤 모두 더합니다.
func ParallelSum(nums []int, parts int) int {
	if len(nums) == 0 {
		return 0
	}
	if parts < 1 {
		parts = 1
	}
	if parts > len(nums) {
		parts = len(nums)
	}

	size := (len(nums) + parts - 1) / parts
	sums := make(chan int)
	started := 0
	for start := 0; start < len(nums); start += size {
		end := min(start+size, len(nums))
		started++
		go func(chunk []int) {
			total := 0
			for _, n := range chunk {
				total += n
			}
			sums <- total
		}(nums[start:end])
	}

	total := 0
	for range started {
		total += <-sums
	}
	return total
}

    문제 2. 가장 빨리 응답한 결과 고르기 (Race)

    mine/exercises 폴더에 아래 테스트를 race_test.go로 입력하고, race.go에 Source 타입과 Race 함수를 구현하세요. 완료 조건: go test ./lesson07/mine/exercises가 통과한다. 여러 소스를 동시에 실행해 가장 먼저 끝난 결과를 돌려주고, timeout 안에 아무도 끝나지 않으면 ctx.Err()를 돌려주며, 늦은 소스에는 취소 신호가 전달되어야 합니다. (Learn Go with Tests select.md의 Racer를 context로 바꾼 문제입니다.)

    package exercises

import (
	"context"
	"errors"
	"testing"
	"time"
)

func after(d time.Duration, answer string, canceled chan<- struct{}) Source {
	return func(ctx context.Context) string {
		select {
		case <-time.After(d):
			return answer
		case <-ctx.Done():
			if canceled != nil {
				canceled <- struct{}{}
			}
			return ""
		}
	}
}

func TestRaceFastestWins(t *testing.T) {
	canceled := make(chan struct{}, 1)
	got, err := Race(context.Background(), time.Second,
		after(300*time.Millisecond, "느림", canceled),
		after(10*time.Millisecond, "빠름", nil),
	)
	if err != nil {
		t.Fatal(err)
	}
	if got != "빠름" {
		t.Errorf("got %q, want %q", got, "빠름")
	}

	select {
	case <-canceled:
	case <-time.After(time.Second):
		t.Error("늦은 작업에 취소 신호가 전달되지 않았습니다")
	}
}

func TestRaceTimeout(t *testing.T) {
	_, err := Race(context.Background(), 30*time.Millisecond,
		after(500*time.Millisecond, "느림1", nil),
		after(500*time.Millisecond, "느림2", nil),
	)
	if !errors.Is(err, context.DeadlineExceeded) {
		t.Errorf("got %v, want context.DeadlineExceeded", err)
	}
}
    힌트 보기

    context.WithTimeout으로 마감이 있는 ctx를 만들고 defer cancel()을 겁니다. 소스마다 고루틴을 시작하고 결과를 소스 개수만큼 버퍼를 가진 채널에 보냅니다(그래야 늦은 소스가 보내다 멈추지 않습니다). 마지막에 select로 "결과 채널"과 ctx.Done() 중 먼저 오는 쪽을 기다립니다. defer cancel()이 늦은 소스의 ctx를 취소해 줍니다.

    정답 예시 보기 
    package exercises

import (
	"context"
	"time"
)

// Source는 결과 문자열 하나를 만드는 작업입니다. ctx가 취소되면 서둘러 돌아와야 합니다.
type Source func(ctx context.Context) string

// Race는 sources를 동시에 실행해 가장 먼저 끝난 결과를 돌려줍니다.
// timeout 안에 아무도 끝나지 않으면 ctx.Err()를 돌려주고, 늦은 작업은 취소됩니다.
func Race(ctx context.Context, timeout time.Duration, sources ...Source) (string, error) {
	ctx, cancel := context.WithTimeout(ctx, timeout)
	defer cancel()

	results := make(chan string, len(sources))
	for _, src := range sources {
		go func() {
			results <- src(ctx)
		}()
	}

	select {
	case r := <-results:
		return r, nil
	case <-ctx.Done():
		return "", ctx.Err()
	}
}

    증상: WaitGroup을 값으로 넘기면 멈추거나 go vet이 경고한다

    sync.WaitGroup을 함수 인자로 값 그대로 넘기면 복사본이 생겨서, 고루틴이 복사본의 Done()을 부르고 원본의 Wait()는 영원히 기다립니다. 실행하면 위와 같은 deadlock이 나고(고루틴 상태가 [sync.WaitGroup.Wait]로 표시됩니다), go vet이 미리 알려 줍니다. 해결은 *sync.WaitGroup(포인터)로 넘기는 것입니다. Mutex도 같은 규칙입니다. 재현하려면 아래 코드를 mine/main.go의 내용으로 바꾸고 터미널(lab 폴더)에서 go vet을 실행합니다. 확인하면 mine/main.go를 다음 실습 내용으로 바꾸거나 정리하기에서 폴더째 지웁니다.

	"fmt"
	"sync"

}

func main() {

		wg.Add(1)
