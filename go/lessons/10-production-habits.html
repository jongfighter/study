    핵심 개념 한국어 요약

    구조화 로그(structured logging). log.Printf("처리 완료 id=%d", id)처럼 문장 속에 값을 섞으면 나중에 기계가 읽기 어렵습니다. slog는 메시지와 key-value 쌍을 따로 받습니다: logger.Info("request", "method", "GET", "status", 200). 같은 호출이 어떤 핸들러(handler)를 쓰느냐에 따라 사람이 읽기 좋은 key=value 텍스트(slog.NewTextHandler)나 한 줄 JSON(slog.NewJSONHandler)으로 출력됩니다. 레벨은 Debug < Info < Warn < Error이고 기본 최소 레벨은 Info입니다. HandlerOptions.Level로 바꿉니다.

    환경변수 설정. 포트, 로그 레벨 같은 실행 환경에 따라 달라지는 값은 코드에 박지 않고 환경변수로 받는 것이 서버 운영의 오랜 관례입니다(Go by Example의 environment-variables 예제가 인용한 12-Factor App의 설정 원칙). 이 코스는 config.Load(getenv) 한 곳에서 읽고, 값이 잘못되면 서버를 시작하기 전에 설명이 있는 에러로 끝냅니다. getenv 함수를 인자로 받아서 테스트는 os.Getenv 대신 맵으로 만든 가짜 환경을 넘깁니다.

    안전한 종료(graceful shutdown). 서버를 그냥 죽이면 처리 중이던 요청이 끊깁니다. 순서는 이렇습니다. ① signal.NotifyContext가 종료 신호(os.Interrupt, SIGTERM)를 받으면 취소되는 context를 줍니다. ② 그 ctx가 취소되면 srv.Shutdown(ctx2)를 부릅니다. Shutdown은 새 연결을 받지 않고, 놀고 있는 연결을 닫고, 진행 중인 요청이 끝나기를 기다립니다. 기다리다가 ctx2가 만료되면 그 에러를 돌려줍니다. ③ Shutdown이 불리면 ListenAndServe/Serve는 즉시 http.ErrServerClosed를 돌려주므로, 프로그램이 그 시점에 끝나지 않고 Shutdown이 돌아올 때까지 기다려야 합니다.(공식 문서) Windows에서는 os.Interrupt(Ctrl+C, Ctrl+Break)만 쓸 수 있고 SIGTERM은 콘솔 닫기·로그오프·시스템 종료 때만 전달되지만, 서버를 Linux 컨테이너에 올리면 종료할 때 SIGTERM이 오므로 둘 다 등록합니다.

    타임아웃과 점검 습관. http.ListenAndServe가 만드는 기본 서버에는 타임아웃이 없어서, 느리게 보내는 클라이언트 하나가 연결을 오래 붙잡을 수 있습니다(100 Go Mistakes #81). 운영 서버는 http.Server를 직접 만들어 ReadHeaderTimeout 등을 정합니다. 코드 점검은 gofmt -l(서식이 다른 파일 목록), go vet(의심스러운 코드), go mod tidy(쓰지 않는 의존성 정리, -diff를 붙이면 파일을 고치지 않고 바뀔 내용만 보여 줌)입니다.

    크로스 컴파일(cross compile). Go는 환경변수 GOOS(운영체제)와 GOARCH(CPU 구조)만 바꿔 다른 플랫폼용 실행 파일을 만들 수 있습니다. 지원 목록은 go tool dist list로 봅니다. 이 코스의 서버는 표준 라이브러리만 쓰므로 별도 런타임 없이 실행 파일 하나로 배포됩니다.

      // Logging은 요청마다 구조화된 로그 한 줄을 남깁니다. 5xx는 ERROR, 4xx는 WARN, 나머지는 INFO입니다.
func Logging(logger *slog.Logger) Middleware {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			start := time.Now()
			rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
			next.ServeHTTP(rec, r)

			level := slog.LevelInfo
			switch {
			case rec.status >= 500:
				level = slog.LevelError
			case rec.status >= 400:
				level = slog.LevelWarn
			}
			logger.Log(r.Context(), level, "request",
				"method", r.Method,
				"path", r.URL.Path,
				"status", rec.status,
				"duration", time.Since(start),
				"request_id", RequestIDFrom(r.Context()),
			)
		})
	}
}

      요청 ID는 RequestID 미들웨어가 만들어 context에 넣습니다. context.WithValue의 키는 다른 패키지의 키와 겹치지 않도록 이 패키지 안에서만 쓰는 타입(requestIDKey)으로 만들었습니다. context에 값을 담는 것은 요청 ID처럼 "정보를 알리는" 용도에 한하고, 함수가 꼭 필요한 값은 타입이 있는 매개변수로 받는 것이 좋습니다(Learn Go with Tests의 context 장).

      // requestIDKey는 context 키로만 쓰는 비공개 타입입니다. 다른 패키지의 키와 겹치지 않습니다.
type requestIDKey struct{}

// RequestIDFrom은 RequestID 미들웨어가 context에 넣어 둔 요청 ID를 꺼냅니다.
func RequestIDFrom(ctx context.Context) string {
	id, _ := ctx.Value(requestIDKey{}).(string)
	return id
}

// RequestID는 요청마다 무작위 ID를 만들어 context와 X-Request-Id 응답 헤더에 넣습니다.
func RequestID(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		buf := make([]byte, 8)
		_, _ = rand.Read(buf)
		id := hex.EncodeToString(buf)

		w.Header().Set("X-Request-Id", id)
		ctx := context.WithValue(r.Context(), requestIDKey{}, id)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
}

      설정 코드는 internal/config/config.go입니다. 기본값을 먼저 채우고, 환경변수가 있으면 검증해서 덮어씁니다.

      func Load(getenv func(string) string) (Config, error) {
	cfg := Config{
		Port:            "8080",
		LogLevel:        slog.LevelInfo,
		LogFormat:       "text",
		ShutdownTimeout: 10 * time.Second,
	}

	if v := getenv("PORT"); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil || n < 1 || n > 65535 {
			return Config{}, fmt.Errorf("PORT는 1~65535 사이의 숫자여야 합니다: %q", v)
		}
		cfg.Port = v
	}

	if v := getenv("LOG_LEVEL"); v != "" {
		if err := cfg.LogLevel.UnmarshalText([]byte(v)); err != nil {
			return Config{}, fmt.Errorf("LOG_LEVEL이 올바르지 않습니다: %q (debug, info, warn, error 중 하나)", v)
		}
	}

	if v := getenv("LOG_FORMAT"); v != "" {
		format := strings.ToLower(v)
		if format != "text" && format != "json" {
			return Config{}, fmt.Errorf("LOG_FORMAT은 text 또는 json이어야 합니다: %q", v)
		}
		cfg.LogFormat = format
	}

	if v := getenv("SHUTDOWN_TIMEOUT"); v != "" {
		d, err := time.ParseDuration(v)
		if err != nil || d <= 0 {
			return Config{}, fmt.Errorf("SHUTDOWN_TIMEOUT은 0보다 큰 시간이어야 합니다(예: 15s): %q", v)
		}
		cfg.ShutdownTimeout = d
	}

	return cfg, nil
}

      // Run은 ln에서 srv를 실행합니다. ctx가 취소되면 새 연결을 받지 않고,
// 진행 중인 요청이 끝나기를 shutdownTimeout까지 기다린 뒤 돌아옵니다.
func Run(ctx context.Context, srv *http.Server, ln net.Listener, shutdownTimeout time.Duration, logger *slog.Logger) error {
	serveErr := make(chan error, 1)
	go func() {
		logger.Info("서버 시작", "addr", ln.Addr().String())
		err := srv.Serve(ln)
		if errors.Is(err, http.ErrServerClosed) {
			err = nil
		}
		serveErr <- err
	}()

	select {
	case err := <-serveErr:
		return err
	case <-ctx.Done():
	}

	logger.Info("종료 신호를 받았습니다. 진행 중인 요청을 마무리합니다", "timeout", shutdownTimeout)

	// ctx는 이미 취소되었으므로 그대로 쓰면 Shutdown이 기다리지 않고 바로 실패합니다.
	shutdownCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), shutdownTimeout)
	defer cancel()

	if err := srv.Shutdown(shutdownCtx); err != nil {
		_ = srv.Close()
		return fmt.Errorf("graceful shutdown 실패: %w", err)
	}
	if err := <-serveErr; err != nil {
		return err
	}
	logger.Info("서버가 정상 종료되었습니다")
	return nil
}

    4. 서버 타임아웃과 점검 명령 세 가지

      internal/app/app.go 위쪽의 NewServer가 기본 서버에 없는 타임아웃을 채웁니다. 헤더를 읽는 데 5초(ReadHeaderTimeout), 요청 전체를 읽는 데 10초, 응답을 쓰는 데 15초, 연결을 놀리는 데 60초가 한도입니다. 값은 예시이며 큰 파일을 내려보내거나 스트리밍하는 서버라면 WriteTimeout을 더 크게 잡아야 합니다. 0은 "제한 없음"입니다.

      // NewServer는 운영에 쓸 만한 타임아웃을 설정한 http.Server를 만듭니다.
// http.ListenAndServe가 쓰는 기본 서버에는 이런 타임아웃이 없습니다.
func NewServer(h http.Handler) *http.Server {
	return &http.Server{
		Handler:           h,
		ReadHeaderTimeout: 5 * time.Second,
		ReadTimeout:       10 * time.Second,
		WriteTimeout:      15 * time.Second,
		IdleTimeout:       60 * time.Second,
	}
}

    8. 연습용 프로필 안에서 keybindings.json에 단축키 규칙 직접 추가하기

      먼저 제목 표시줄이나 관리 버튼에서 go-practice 프로필을 쓰는 중인지 확인합니다. Ctrl+K Ctrl+S로 키보드 바로 가기(Keyboard Shortcuts) 편집기를 엽니다. 위쪽 검색창에 Run Task를 입력하면 명령 Tasks: Run Task가 보입니다. 이 명령에 단축키를 붙이면 매번 목록에서 작업을 골라야 하므로, 아래에서는 어떤 작업을 실행할지까지 JSON으로 정합니다. 편집기 제목 줄 오른쪽의 키보드 바로 가기 열기(JSON)(Open Keyboard Shortcuts (JSON)) 버튼을 눌러 keybindings.json을 열고 아래 규칙을 넣습니다. 공식 문서는 키보드 단축키가 프로필의 구성 요소라고 설명하므로, 프로필을 쓰는 동안 이 규칙은 연습용 프로필에 기록되어야 합니다. 문서에는 프로필 안의 keybindings.json 위치가 적혀 있지 않으니, 파일 경로를 추측하는 대신 10단계에서 기본 프로필로 돌아가 같은 키를 눌러 아무 일도 없는지로 확인합니다.

      [
  {
    "key": "ctrl+alt+t",
    "command": "workbench.action.tasks.runTask",
    "args": "go: run task api"
  }
]

      이 규칙은 "ctrl+alt+t를 누르면 workbench.action.tasks.runTask 명령을 go: run task api라는 인자와 함께 실행하라"는 뜻입니다. 이미 파일에 [ ... ]가 있다면 그 안에 항목만 추가합니다. 저장하면 바로 적용됩니다(문서: keybindings.json은 VS Code가 감시하고 있어 고치면 즉시 반영). 같은 키에 이미 다른 명령이 연결되어 있지 않은지 먼저 확인하세요: 키보드 바로 가기 편집기 검색창에서 키 입력 기록 버튼을 눌러 그 키를 직접 눌러 보면 그 키에 연결된 명령이 나옵니다. 항목을 우클릭해 같은 키 바인딩 표시(Show Same Keybindings)를 고를 수도 있습니다. 이미 쓰이는 조합이면 다른 키로 바꾸세요. 확인: 코드 편집기를 클릭하고 방금 정한 키를 누르면 go: run task api가 서버 전용 터미널에서 시작됩니다. 서버는 Ctrl+C로 끝냅니다.

      이 규칙은 연습용 프로필과 함께 지울 것이라 따로 되돌리지 않습니다. 규칙 앞에 -를 붙인 "command": "-명령" 형태는 기본 단축키를 지우는 문법입니다.

    9. 연습용 프로필 안에서 설정 JSON 고치기: 사용자 값과 워크스페이스 값의 우선순위

      ② 설정 화면에서 Workspace 탭(한국어 UI의 작업 영역)을 선택하고 editor.rulers의 Edit in settings.json을 누르면 이 코스의 .vscode/settings.json이 열립니다(files.eol, [go] 묶음, gopls 설정이 이미 들어 있습니다. 링크가 없으면 Preferences: Open Workspace Settings (JSON)). 새로 들어간 "editor.rulers": []를 "editor.rulers": [60]으로 고쳐 저장합니다. 이것은 프로필이 아니라 코스 폴더의 파일을 고치는 것이므로 ④에서 바로 되돌립니다.

      ③ 확인: 이제 세로선은 60 위치에만 있고 100 위치의 선은 사라집니다. VS Code 문서에서 설정은 "기본값 → 사용자 → 워크스페이스" 순서로 덮어쓰고(나중 것이 이긴다), 숫자·문자·배열 값은 합쳐지지 않고 통째로 바뀌기 때문입니다. 반대로 search.exclude처럼 객체 값은 합쳐집니다(레슨 09의 설정 체험). 설정 화면의 User와 Workspace 탭에서 editor.rulers를 보면 값이 따로 표시되고 바뀐 항목에는 표시가 붙습니다.

      ④ 확인이 끝났으니 지금 바로 .vscode/settings.json을 원래대로 되돌립니다. 방금 넣은 editor.rulers 줄만 지우고(그 줄에 커서를 두고 Ctrl+Shift+K) 저장합니다. 복원 확인: ㉠ 이 파일에서 Ctrl+F로 editor.rulers를 찾으면 결과가 없습니다. ㉡ 세로선이 다시 100 위치에만 보입니다(연습용 프로필의 사용자 값이 남아 있는 동안). ㉢ 설정 화면 Workspace 탭의 @modified 검색 결과에 editor.rulers가 없습니다.

    lab/lesson10/mine/exercises 폴더를 만들고 아래 테스트를 maxbody_test.go로 입력한 뒤(첫 줄은 package exercises 그대로), maxbody.go에 MaxBody(limit int64) func(http.Handler) http.Handler와 IsTooLarge(err error) bool을 구현하세요. 완료 조건: go test ./lesson10/mine/exercises가 통과한다. 요청이 보낸 Content-Length가 한도보다 크면 다음 핸들러를 부르지 않고 413과 {"error":"요청 본문이 너무 큽니다"}를 돌려주고, 길이를 모르는(chunked) 요청은 본문을 읽는 도중에 한도를 넘으면 읽기 에러가 나게 합니다. 한도 없는 본문은 서버의 메모리를 마음대로 쓰게 해 주는 것이라 운영 서버의 흔한 보강 항목입니다.

    package exercises

import (
	"io"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"
)

func TestMaxBodyRejectsByContentLength(t *testing.T) {
	called := false
	next := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		called = true
	})
	req := httptest.NewRequest(http.MethodPost, "/tasks", strings.NewReader(strings.Repeat("x", 100)))
	rec := httptest.NewRecorder()

	MaxBody(10)(next).ServeHTTP(rec, req)

	if rec.Code != http.StatusRequestEntityTooLarge {
		t.Errorf("status = %d, want 413", rec.Code)
	}
	if called {
		t.Error("제한을 넘으면 다음 핸들러를 부르면 안 됩니다")
	}
}

func TestMaxBodyCutsUnknownLengthBody(t *testing.T) {
	var readErr error
	next := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		_, readErr = io.ReadAll(r.Body)
	})
	req := httptest.NewRequest(http.MethodPost, "/tasks", strings.NewReader(strings.Repeat("x", 100)))
	req.ContentLength = -1

	MaxBody(10)(next).ServeHTTP(httptest.NewRecorder(), req)

	if !IsTooLarge(readErr) {
		t.Errorf("읽는 도중 MaxBytesError가 나와야 합니다: %v", readErr)
	}
}

func TestMaxBodyAllowsSmallBody(t *testing.T) {
	var got string
	next := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		b, _ := io.ReadAll(r.Body)
		got = string(b)
	})
	req := httptest.NewRequest(http.MethodPost, "/tasks", strings.NewReader("작은 본문"))

	MaxBody(1024)(next).ServeHTTP(httptest.NewRecorder(), req)

	if got != "작은 본문" {
		t.Errorf("got %q", got)
	}
}
    힌트 보기

    r.ContentLength는 길이를 모르면 -1입니다. 한도와 비교해 크면 바로 응답하고 return합니다(return을 잊지 마세요). 그렇지 않으면 r.Body = http.MaxBytesReader(w, r.Body, limit)로 본문을 감싼 뒤 next.ServeHTTP(w, r)를 부릅니다. 한도를 넘게 읽으려 할 때 돌아오는 에러의 타입은 *http.MaxBytesError이고, 이런 에러 타입은 errors.As로 확인합니다(레슨 05).

    정답 예시 보기 
    // Package exercises는 레슨 10 연습 문제의 정답 예시입니다.
package exercises

import (
	"errors"
	"net/http"
)

// MaxBody는 요청 본문을 limit바이트까지만 읽도록 제한합니다.
// Content-Length가 이미 limit보다 크면 핸들러를 부르지 않고 413으로 응답하고,
// 길이를 모르는 요청은 http.MaxBytesReader로 감싸 읽는 도중에 끊습니다.
func MaxBody(limit int64) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if r.ContentLength > limit {
				w.Header().Set("Content-Type", "application/json")
				w.WriteHeader(http.StatusRequestEntityTooLarge)
				_, _ = w.Write([]byte(`{"error":"요청 본문이 너무 큽니다"}` + "\n"))
				return
			}
			r.Body = http.MaxBytesReader(w, r.Body, limit)
			next.ServeHTTP(w, r)
		})
	}
}

// IsTooLarge는 body를 읽다가 난 에러가 크기 제한 때문인지 알려 줍니다.
func IsTooLarge(err error) bool {
	var tooLarge *http.MaxBytesError
	return errors.As(err, &tooLarge)
}

    문제 2. 로그에서 비밀 값 가리기

    mine/exercises에 mask.go를 만들어, slog.HandlerOptions.ReplaceAttr에 넘길 수 있는 함수 MaskSecrets(groups []string, a slog.Attr) slog.Attr를 구현하세요. 키가 password, token, authorization(대소문자 무시)인 속성의 값은 ***로 바꾸고, 다른 속성은 그대로 둡니다. 완료 조건: 아래 테스트를 mask_test.go로 입력해 go test ./lesson10/mine/exercises가 통과한다. 로그에 비밀번호가 남으면 로그 파일이 곧 유출 지점이 됩니다.

    package exercises

import (
	"bytes"
	"log/slog"
	"strings"
	"testing"
)

func newMaskedLogger(buf *bytes.Buffer) *slog.Logger {
	return slog.New(slog.NewTextHandler(buf, &slog.HandlerOptions{ReplaceAttr: MaskSecrets}))
}

func TestMaskSecretsHidesSensitiveValues(t *testing.T) {
	var buf bytes.Buffer
	logger := newMaskedLogger(&buf)

	logger.Info("로그인 시도", "user", "gopher", "Password", "p@ssw0rd", "token", "abc123")

	out := buf.String()
	for _, secret := range []string{"p@ssw0rd", "abc123"} {
		if strings.Contains(out, secret) {
			t.Errorf("로그에 비밀 값 %q가 그대로 남았습니다: %s", secret, out)
		}
	}
	for _, want := range []string{"user=gopher", "Password=***", "token=***"} {
		if !strings.Contains(out, want) {
			t.Errorf("로그에 %q가 있어야 합니다: %s", want, out)
		}
	}
}

func TestMaskSecretsKeepsOtherAttributes(t *testing.T) {
	var buf bytes.Buffer
	logger := newMaskedLogger(&buf)

	logger.Warn("요청", "path", "/tasks", "status", 404)

	out := buf.String()
	if !strings.Contains(out, "path=/tasks") || !strings.Contains(out, "status=404") {
		t.Errorf("다른 속성은 그대로여야 합니다: %s", out)
	}
}
    힌트 보기

    ReplaceAttr는 로그 한 속성이 출력되기 직전에 불리는 함수이고, 돌려준 slog.Attr가 대신 출력됩니다. 키를 소문자로 바꿔(strings.ToLower) 비밀 키 목록(맵)에 있는지 보고, 있으면 slog.String(a.Key, "***")을, 아니면 받은 a를 그대로 돌려줍니다. 레슨 10의 middleware_test.go에서 시간 속성을 지우는 newBufferLogger도 같은 ReplaceAttr를 씁니다.

    정답 예시 보기 
    package exercises

import (
	"log/slog"
	"strings"
)

var secretKeys = map[string]bool{
	"password":      true,
	"token":         true,
	"authorization": true,
}

// MaskSecrets는 slog.HandlerOptions.ReplaceAttr에 넘겨 쓰는 함수입니다.
// 키가 password, token, authorization(대소문자 무시)인 속성의 값을 ***로 바꿉니다.
func MaskSecrets(groups []string, a slog.Attr) slog.Attr {
	if secretKeys[strings.ToLower(a.Key)] {
		return slog.String(a.Key, "***")
	}
	return a
}
