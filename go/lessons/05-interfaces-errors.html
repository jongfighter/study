    인터페이스와 에러 처리

    선언 없이 만족하는 인터페이스, 상속 없이 기능을 합치는 임베딩, 예외 대신 값으로 다루는 error와 errors.Is/errors.As를 코드로 확인합니다. 컴파일 오류를 문제 패널과 빠른 수정으로 고치는 방법도 연습합니다.

    • 난이도 ★★★☆☆
    • 선행 레슨 04

    GOAL학습 목표

    • 인터페이스를 정의하고, implements 선언 없이 메서드만 갖춘 타입이 인터페이스를 만족하는 것을 컴파일러로 확인할 수 있다.
    • 구조체 임베딩으로 다른 타입의 메서드를 승격(promote)받는 코드를 작성하고, 상속과 어떻게 다른지 설명할 수 있다.
    • fmt.Errorf("...: %w", err)로 오류를 감싸고 errors.Is(값 비교)와 errors.As(타입 비교)로 구분할 수 있다.
    • defer의 실행 순서와 panic/recover를 설명하고, panic을 일반 오류 처리에 쓰지 않는 이유를 말할 수 있다.
    • 문제 패널, F8, 빠른 수정(Ctrl+.), 호버, 구현으로 이동(Ctrl+F12)으로 컴파일 오류를 읽고 고칠 수 있다.

    핵심 개념 한국어 요약

    인터페이스(interface)의 암묵적 구현. 인터페이스는 메서드 시그니처의 모음입니다. Kotlin처럼 class X : Notifier라고 선언하지 않고, 타입이 인터페이스의 메서드를 모두 갖기만 하면 자동으로 만족합니다. 그래서 인터페이스는 사용하는 쪽에서 필요한 만큼만 작게 정의하는 것이 Go의 관례입니다(100 Go Mistakes #5, #6). 표준 라이브러리의 fmt.Stringer(String() string 메서드 하나)를 구현하면 fmt.Println이 그 메서드로 출력합니다. any는 메서드가 하나도 없는 인터페이스라 모든 값을 담을 수 있고, value.(타입)이나 switch v := value.(type)으로 실제 타입을 꺼냅니다.

    임베딩(embedding). 필드 이름 없이 타입만 적으면 그 타입의 필드와 메서드가 바깥 구조체로 승격됩니다. 상속이 아니라 조합이라, 바깥 타입이 안쪽 타입의 하위 타입이 되지는 않습니다. 인터페이스를 임베딩하면 바깥 구조체도 그 인터페이스를 만족합니다.

    error 값. error는 Error() string 메서드 하나뿐인 내장 인터페이스입니다. 실패할 수 있는 함수는 마지막 반환값으로 error를 돌려주고, 호출자가 if err != nil로 확인합니다. 호출자가 오류 종류를 구분해야 하면 미리 만든 값(sentinel error, var ErrNotFound = errors.New(...))이나 커스텀 오류 타입을 씁니다. fmt.Errorf("...: %w", err)의 %w는 원래 오류를 감싸서 맥락을 덧붙이고, 감싼 뒤에는 == 비교가 실패하므로 errors.Is(err, ErrNotFound)(감싼 사슬을 풀며 값 비교)와 errors.As(err, &대상)(사슬에서 해당 타입 찾기)를 씁니다.

    defer, panic, recover. defer는 함수가 끝날 때 실행할 호출을 예약하며 나중에 예약한 것이 먼저(LIFO) 실행됩니다. panic은 프로그램 흐름을 중단시키고, defer 안의 recover()가 그것을 붙잡아 일반 오류로 바꿀 수 있습니다. panic은 프로그래머의 실수나 시작 시점의 치명적 실패처럼 복구할 수 없는 경우에만 쓰고, 일반적인 실패는 error로 돌려주는 것이 Go의 원칙입니다(100 Go Mistakes #48). 오류는 한 번만 처리해야 하므로(로그를 남기거나 돌려주거나 둘 중 하나, #52) 감싸서 위로 돌려주는 방식을 흔히 씁니다.

    1. 인터페이스 프로그램 입력하고 실행하기

      탐색기(Ctrl+Shift+E)에서 lab/lesson05를 우클릭해 새 폴더 mine/interfaces를 만들고 그 안에 main.go를 만들어 아래 코드를 직접 입력합니다. Notifier를 구현했다는 선언이 어디에도 없다는 점을 눈여겨보세요.

      package main

import "fmt"

type Notifier interface {
	Notify(message string) string
}

type EmailNotifier struct {
	Address string
}

func (e EmailNotifier) Notify(message string) string {
	return "이메일(" + e.Address + "): " + message
}

type ConsoleNotifier struct{}

func (ConsoleNotifier) Notify(message string) string {
	return "콘솔: " + message
}

type Logger struct {
	Prefix string
}

func (l Logger) Log(message string) {
	fmt.Println(l.Prefix + message)
}

type Service struct {
	Logger
	Notifier
}

type Task struct {
	ID    int
	Title string
}

func (t Task) String() string {
	return fmt.Sprintf("Task#%d(%s)", t.ID, t.Title)
}

func broadcast(notifiers []Notifier, message string) {
	for _, n := range notifiers {
		fmt.Println(n.Notify(message))
	}
}

func describe(value any) string {
	switch v := value.(type) {
	case EmailNotifier:
		return "이메일 알림, 주소=" + v.Address
	case ConsoleNotifier:
		return "콘솔 알림"
	case nil:
		return "nil"
	default:
		return fmt.Sprintf("알 수 없는 타입 %T", v)
	}
}

func main() {
	notifiers := []Notifier{
		EmailNotifier{Address: "dev@example.com"},
		ConsoleNotifier{},
	}
	broadcast(notifiers, "배포가 끝났습니다")

	service := Service{
		Logger:   Logger{Prefix: "[service] "},
		Notifier: ConsoleNotifier{},
	}
	service.Log("시작")
	fmt.Println(service.Notify("임베딩으로 메서드를 물려받습니다"))

	fmt.Println(Task{ID: 1, Title: "인터페이스 익히기"})

	fmt.Println(describe(EmailNotifier{Address: "a@b.c"}))
	fmt.Println(describe(ConsoleNotifier{}))
	fmt.Println(describe(42))
}

    4. 빠른 수정으로 메서드 뼈대 만들기

      3단계에서 오류가 난 SlackNotifier{} 위에 커서를 놓고 Ctrl+.(빠른 수정)를 누릅니다. 목록에서 Declare missing methods of Notifier를 고르고 엔터 키를 누릅니다(문구는 gopls 버전에 따라 조금 다를 수 있습니다). gopls가 SlackNotifier 아래에 아래와 같은 메서드 뼈대를 만들어 줍니다.

      // Notify implements [Notifier].

	panic("unimplemented")

      본문의 panic("unimplemented")를 return "슬랙: " + message로 바꾸고 저장한 뒤 다시 실행하면 오류가 사라지고 출력 맨 끝에 슬랙: 테스트가 추가됩니다. 이 코드가 처음 만들어진 뼈대처럼 panic으로 남아 있으면 실행할 때 프로그램이 멈추니 반드시 채워 주세요. 이 실험이 끝나면 SlackNotifier 관련 코드는 지워도 됩니다.

      다른 빠른 수정도 하나 써 봅니다. main 함수의 맨 아래에 쓰지 않을 변수 count := 3을 추가하면 declared and not used: count 오류가 납니다. 이 변수 위에서 Ctrl+.를 누르면 Remove variable count가 나오고 고르면 그 줄이 지워집니다(gopls로 확인한 항목입니다).

    5. 오류 값 프로그램 입력하고 실행하기

      lab/lesson05/mine 아래에 errors 폴더와 main.go를 만들어 입력합니다.

      package main

import (
	"errors"
	"fmt"
	"strings"
)

var ErrNotFound = errors.New("할 일을 찾을 수 없습니다")

type ValidationError struct {
	Field  string
	Reason string
}

func (e *ValidationError) Error() string {
	return e.Field + ": " + e.Reason
}

type Store struct {
	tasks map[int]string
}

func NewStore() *Store {
	return &Store{tasks: make(map[int]string)}
}

func (s *Store) Add(id int, title string) error {
	if strings.TrimSpace(title) == "" {
		return &ValidationError{Field: "title", Reason: "비어 있을 수 없습니다"}
	}
	s.tasks[id] = title
	return nil
}

func (s *Store) Get(id int) (string, error) {
	title, ok := s.tasks[id]
	if !ok {
		return "", fmt.Errorf("할 일 %d 조회: %w", id, ErrNotFound)
	}
	return title, nil
}

func main() {
	store := NewStore()

	if err := store.Add(1, "에러 처리 배우기"); err != nil {
		fmt.Println("추가 실패:", err)
	}

	title, err := store.Get(1)
	fmt.Println("조회:", title, err)

	_, err = store.Get(99)
	fmt.Println("오류:", err)
	fmt.Println("== 로 비교:", err == ErrNotFound)
	fmt.Println("errors.Is:", errors.Is(err, ErrNotFound))
	fmt.Println("Unwrap:", errors.Unwrap(err))

	err = store.Add(2, "   ")
	wrapped := fmt.Errorf("할 일 추가: %w", err)
	fmt.Println("오류:", wrapped)

	var validationErr *ValidationError
	if errors.As(wrapped, &validationErr) {
		fmt.Println("검증 실패 필드:", validationErr.Field)
	}

	switch {
	case errors.Is(wrapped, ErrNotFound):
		fmt.Println("404로 응답")
	case errors.As(wrapped, &validationErr):
		fmt.Println("400으로 응답")
	default:
		fmt.Println("500으로 응답")
	}
}

      핵심은 4번째와 5번째 줄입니다. 감싼 오류(할 일 99 조회: ...)는 == ErrNotFound로 비교하면 false이지만, errors.Is는 감싼 사슬을 풀어 보고 true를 돌려줍니다. 7~8번째 줄에서는 errors.As가 감싼 오류 안에서 *ValidationError를 찾아 Field를 꺼냈습니다. 마지막 switch는 이 오류를 HTTP 상태 코드로 바꾸는 방식의 축소판이고, 레슨 08에서 실제로 씁니다. 직접 확인해 보려면 fmt.Println("errors.Is:", errors.Is(err, ErrNotFound)) 줄의 errors.Is(err, ErrNotFound)를 err == ErrNotFound로 바꿔 실행하세요. 그 줄이 errors.Is: false로 바뀌고 나머지 줄은 그대로입니다. 확인한 뒤 원래대로 되돌립니다.

    6. defer와 recover 프로그램 입력하고 실행하기

      recovery 폴더와 main.go를 만들어 입력합니다.

      package main

import "fmt"

func deferOrder() {
	for i := 1; i <= 3; i++ {
		defer fmt.Println("defer", i)
	}
	fmt.Println("본문 끝")
}

func safeDivide(a, b int) (result int, err error) {
	defer func() {
		if r := recover(); r != nil {
			err = fmt.Errorf("복구했습니다: %v", r)
		}
	}()
	return a / b, nil
}

func main() {
	deferOrder()

	result, err := safeDivide(10, 2)
	fmt.Println(result, err)

	result, err = safeDivide(1, 0)
	fmt.Println(result, err)

	fmt.Println("프로그램은 계속 실행됩니다")
}

프로그램은 계속 실행됩니다

      defer가 defer 3, defer 2, defer 1 순서로 실행되었습니다. safeDivide(1, 0)은 정수를 0으로 나누는 런타임 패닉을 일으키지만, defer 안의 recover()가 그 패닉을 붙잡아 오류로 바꿨습니다. 그래서 마지막 줄이 출력됩니다. 반환값을 (result int, err error)처럼 이름 붙여 둔 이유는 defer 안에서 err를 바꿔야 하기 때문입니다.

    문제 1. 도형 인터페이스

    Area() float64 메서드 하나를 가진 Shape 인터페이스와 이를 만족하는 Rectangle(가로, 세로), Circle(반지름)을 만들고, []Shape의 넓이 합을 돌려주는 totalArea를 작성하세요. 이 문제는 Learn Go with Tests의 구조체·메서드·인터페이스 장과 같은 주제입니다. 완료 조건: 가로 3, 세로 4인 사각형과 반지름 1인 원의 합이 도형 넓이 합: 15.14로 출력된다(%.2f로 출력). implements에 해당하는 선언을 쓰지 않는다.

    힌트 보기

    원주율은 math.Pi입니다(import "math"). 두 타입 모두 func (r Rectangle) Area() float64처럼 값 수신자로 만들면 Rectangle{Width: 3, Height: 4} 그대로 Shape 슬라이스에 넣을 수 있습니다. totalArea는 슬라이스를 돌며 shape.Area()를 더합니다.

    정답 예시 보기 
    type Shape interface {
	Area() float64
}

type Rectangle struct {
	Width, Height float64
}

func (r Rectangle) Area() float64 {
	return r.Width * r.Height
}

type Circle struct {
	Radius float64
}

func (c Circle) Area() float64 {
	return math.Pi * c.Radius * c.Radius
}

func totalArea(shapes []Shape) float64 {
	total := 0.0
	for _, shape := range shapes {
		total += shape.Area()
	}
	return total
}

    문제 2. 포트 번호 파싱과 오류 감싸기

    문자열을 포트 번호로 바꾸는 parsePort(text string) (int, error)를 작성하세요. 숫자가 아니거나 1~65535 범위를 벗어나면 미리 만든 ErrInvalidPort를 %w로 감싸 돌려줍니다. 완료 조건: "8080"은 성공이고, "abc"와 "70000"은 errors.Is(err, ErrInvalidPort)가 true이다.

    힌트 보기

    문자열을 정수로 바꾸는 함수는 strconv.Atoi입니다((int, error) 반환). 그 오류를 그대로 돌려주지 말고 fmt.Errorf("포트 %q: %w", text, ErrInvalidPort)처럼 ErrInvalidPort를 감싸야 errors.Is가 통과합니다. 범위 검사는 if port < 1 || port > 65535입니다.

    정답 예시 보기 
    var ErrInvalidPort = errors.New("잘못된 포트 번호")

func parsePort(text string) (int, error) {
	port, err := strconv.Atoi(text)
	if err != nil {
		return 0, fmt.Errorf("포트 %q: %w", text, ErrInvalidPort)
	}
	if port < 1 || port > 65535 {
		return 0, fmt.Errorf("포트 %d: %w", port, ErrInvalidPort)
	}
	return port, nil
}
