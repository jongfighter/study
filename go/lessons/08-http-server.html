    핵심 개념 한국어 요약

    핸들러(handler). HTTP 요청을 처리하는 것은 http.Handler 인터페이스(메서드 ServeHTTP(w, r) 하나)이고, 보통은 func(w http.ResponseWriter, r *http.Request) 모양의 평범한 함수를 씁니다. w로 응답을 쓰고 r에서 요청을 읽습니다. 애너테이션도 리플렉션 마법도 없는 그냥 함수라서 테스트하기도 쉽습니다(레슨 09).

    ServeMux와 패턴. ServeMux는 요청 URL을 등록된 패턴과 맞춰 알맞은 핸들러를 부르는 라우터입니다. Go 1.22부터 패턴 모양은 [메서드 ][호스트]/[경로]이고, 경로에는 {id}처럼 이름 붙은 와일드카드를 쓸 수 있습니다. 메서드를 적으면 그 메서드만 맞는데, GET은 HEAD도 함께 받습니다. 여러 패턴이 맞으면 "더 구체적인 쪽"이 이기고, 어느 쪽이 더 구체적인지 가릴 수 없으면 등록할 때 패닉이 납니다. 경로는 맞지만 메서드가 다르면 ServeMux가 알아서 405 Method Not Allowed와 Allow 헤더를 보냅니다. 이 코스의 lab/go.mod는 go 1.24라서 이 문법을 그대로 씁니다.

    JSON. encoding/json은 구조체의 대문자로 시작하는(공개) 필드만 다룹니다. 이름은 구조체 태그 `json:"created_at"`로 정합니다. 요청은 json.NewDecoder(r.Body).Decode(&v)로 읽고, 응답은 json.NewEncoder(w).Encode(v)로 씁니다. DisallowUnknownFields()를 켜면 모르는 필드가 들어왔을 때 에러가 납니다. 이 레슨의 JSON은 널리 쓰는 encoding/json입니다. 클론한 Go by Example의 json 예제는 encoding/json/v2를 가져오는데, 이 코스에서는 쓰지 않습니다.

    응답 쓰기 순서. 응답은 상태 코드 → 헤더 → 본문 순서로 나갑니다. w.WriteHeader(201)은 헤더를 확정하므로 헤더(Content-Type 등)는 그 전에 설정해야 하고, 본문을 한 번이라도 썼다면 상태 코드는 자동으로 200이 됩니다. http.Error나 writeError를 부른 뒤에는 return으로 핸들러를 끝내야 합니다(아래 "자주 나는 오류"에서 실제로 재현합니다).

    요청마다 고루틴. net/http 서버는 요청 하나를 고루틴 하나로 처리합니다. 그래서 여러 요청이 같은 저장소(맵)를 동시에 건드립니다. 레슨 07의 Mutex가 여기서 필요합니다.

    1. 가장 작은 서버 만들고 터미널 나누기

      탐색기(Ctrl+Shift+E)에서 lab/lesson08 폴더에 새 폴더 mine을 만들고, 그 안에 새 폴더 hello, 다시 그 안에 새 파일 main.go를 만들어 아래 코드를 입력합니다. "GET /hello/{name}"의 {name}이 경로 변수이고 r.PathValue("name")으로 읽습니다.

      package main

import (
	"fmt"
	"log"
	"net/http"
)

func main() {
	mux := http.NewServeMux()

	mux.HandleFunc("GET /health", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintln(w, "ok")
	})
	mux.HandleFunc("GET /hello/{name}", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintf(w, "안녕하세요, %s!\n", r.PathValue("name"))
	})

	log.Println("서버 시작: http://localhost:8080")
	log.Fatal(http.ListenAndServe(":8080", mux))
}

      • Microsoft 문서의 Windows 방화벽 규칙: 관리자 권한이 있는 사용자가 창을 취소하거나 아니오로 답하면 그 프로그램의 차단 규칙이 만들어지고, 관리자가 아닌 사용자는 어느 버튼을 눌러도 차단 규칙이 만들어집니다. 규칙은 지우기 전까지 남고 같은 프로그램에는 창이 다시 뜨지 않습니다. 이 문서는 같은 PC 안(loopback) 접속이 영향을 받는지는 말하지 않습니다.
      • 같은 PC 안에서 일반 프로그램끼리 하는 loopback 접속은 Windows가 기본으로 허용하는 필터가 따로 있다는 분석이 있습니다(Google Project Zero의 2021년 분석 글). 그래서 일반적으로 취소해도 localhost 접속은 됩니다. 다만 이 PC에서 방화벽 창을 실제로 띄워 확인하지는 못했습니다(관리자 권한이 필요한 조작이라 실행하지 않았습니다).
      • go run은 실행할 때마다 임시 폴더에 새 실행 파일을 만듭니다. 방화벽 규칙이 프로그램 경로를 기준으로 만들어진다면 같은 코드라도 실행할 때마다 창이 다시 뜰 수 있습니다(이 PC에서는 확인하지 못했습니다). 반복돼도 같은 방법으로 취소하면 됩니다.
      • go test가 쓰는 httptest는 127.0.0.1에서만 서버를 열어서(Go 소스의 net/http/httptest/server.go에서 확인) 이 창이 뜨지 않습니다.

    3. 목록 조회와 생성 구현하기 (JSON 읽고 쓰기, 검증)

      mine/main.go에서 Ctrl+F로 func (s *server) listTasks를 찾아 501을 돌려주던 함수를 아래로 바꿉니다. 목록은 저장소가 돌려준 슬라이스를 그대로 JSON 배열로 씁니다. 저장소의 list는 make([]Task, 0, ...)로 빈 슬라이스를 만들기 때문에 할 일이 하나도 없어도 []가 나갑니다(nil 슬라이스였다면 null이 나가는 문제는 "자주 나는 오류"에서 봅니다).

      func (s *server) listTasks(w http.ResponseWriter, r *http.Request) {

    5. 수정과 삭제 구현하기 (포인터 필드, 204)

      updateTask는 일부 필드만 바꾸는 PATCH입니다. 요청 구조체의 필드를 *string, *bool 같은 포인터로 두면 "보내지 않음"은 nil이고, 보냈다면 값이 있는 포인터가 됩니다. "done": false와 "보내지 않음"을 구분하려고 bool 대신 *bool을 씁니다. 아래 코드로 updateTask 스텁을 바꿉니다.

      type updateTaskRequest struct {
	Title *string `json:"title"`
	Done  *bool   `json:"done"`
}

func (s *server) updateTask(w http.ResponseWriter, r *http.Request) {
	id, err := parseID(r)
	if err != nil {
		writeError(w, http.StatusBadRequest, err.Error())
		return
	}
	var req updateTaskRequest
	if err := decodeJSON(r, &req); err != nil {
		writeError(w, http.StatusBadRequest, "요청 본문을 읽을 수 없습니다: "+err.Error())
		return
	}
	if req.Title == nil && req.Done == nil {
		writeError(w, http.StatusBadRequest, "수정할 필드(title, done)가 없습니다")
		return
	}
	if req.Title != nil {
		title := strings.TrimSpace(*req.Title)
		if title == "" {
			writeError(w, http.StatusBadRequest, "title은 비어 있을 수 없습니다")
			return
		}
		req.Title = &title
	}

	task, ok := s.store.update(id, req.Title, req.Done)
	if !ok {
		writeError(w, http.StatusNotFound, "할 일을 찾을 수 없습니다")
		return
	}
	writeJSON(w, http.StatusOK, task)
}

package exercises

import (
	"errors"
	"strings"
	"unicode/utf8"
)

const maxTitleRunes = 100

// ValidateTitle은 제목의 앞뒤 공백을 지우고, 비어 있거나 100글자(룬 기준)를 넘으면 에러를 돌려줍니다.
// len(title)은 글자 수가 아니라 바이트 수이므로 utf8.RuneCountInString으로 셉니다.
func ValidateTitle(title string) (string, error) {
	title = strings.TrimSpace(title)
	if title == "" {
		return "", errors.New("title은 비어 있을 수 없습니다")
	}
	if utf8.RuneCountInString(title) > maxTitleRunes {
		return "", errors.New("title은 100자 이하여야 합니다")
	}
	return title, nil
}

    문제 2. 이미 완료된 할 일만 지우는 엔드포인트 규칙 만들기

    삭제 규칙을 바꿉니다. 완료(done: true)된 할 일만 삭제할 수 있고, 완료되지 않은 할 일을 지우려 하면 409 Conflict와 {"error":"완료된 할 일만 삭제할 수 있습니다"}를 돌려주세요. 완료 조건: 할 일을 만들고 바로 DELETE하면 409, PATCH로 done을 true로 바꾼 뒤 DELETE하면 204가 나오고, 없는 id는 여전히 404다.

    힌트 보기

    저장소의 remove가 한 번에 "확인하고 지우기"를 하면 락 안에서 안전합니다. 지금 remove는 bool 하나만 돌려주므로, 결과를 구분할 수 있게 (지웠음 / 없음 / 완료 전) 세 가지 중 하나를 돌려주도록 반환 값을 바꾸세요. 열거형이 없는 Go에서는 error 값(errNotFound, errNotDone)을 돌려주고 핸들러에서 errors.Is로 구분하는 방식이 자연스럽습니다. 핸들러는 errors.Is 결과에 따라 404, 409를 고릅니다.

    정답 예시 보기 
    package exercises

import (
	"errors"
	"sync"
)

var (
	// ErrNotFound는 해당 id의 할 일이 없을 때의 에러입니다. 핸들러에서 404로 바꿉니다.
	ErrNotFound = errors.New("할 일을 찾을 수 없습니다")
	// ErrNotDone은 완료되지 않은 할 일을 지우려 할 때의 에러입니다. 핸들러에서 409로 바꿉니다.
	ErrNotDone = errors.New("완료된 할 일만 삭제할 수 있습니다")
)

// Store는 id와 완료 여부만 가진 아주 작은 저장소입니다.
type Store struct {
	mu    sync.Mutex
	items map[int]bool
}

// NewStore는 빈 저장소를 만듭니다.
func NewStore() *Store {
	return &Store{items: make(map[int]bool)}
}

// Add는 id에 해당하는 할 일을 완료 여부와 함께 저장합니다.
func (s *Store) Add(id int, done bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.items[id] = done
}

// Remove는 완료된 할 일만 지웁니다. 확인과 삭제를 같은 락 안에서 하므로
// 그 사이에 다른 요청이 끼어들 수 없습니다.
func (s *Store) Remove(id int) error {
	s.mu.Lock()
	defer s.mu.Unlock()

	done, ok := s.items[id]
	switch {
	case !ok:
		return ErrNotFound
	case !done:
		return ErrNotDone
	}
	delete(s.items, id)
	return nil
}
