    • var와 :=를 상황에 맞게 골라 쓰고, 타입을 쓰지 않아도 추론되는 타입을 %T로 확인할 수 있다.
    • 타입별 zero value(0, "", false, nil)를 말하고, 명시적 타입 변환이 필요한 이유를 설명할 수 있다.

    핵심 개념 한국어 요약

    변수 선언. var 이름 타입 = 값이 정식 문법이고, 함수 안에서는 이름 := 값(짧은 선언, short variable declaration)을 주로 씁니다. :=는 타입을 값에서 추론하며 함수 밖에서는 쓸 수 없습니다. 선언한 변수를 쓰지 않으면 컴파일 오류입니다.

    zero value. Go에는 초기화하지 않은 변수가 없습니다. 값을 주지 않으면 타입별 기본값이 들어갑니다: 숫자 0, 문자열 "", 불리언 false, 포인터·슬라이스·맵·함수·인터페이스는 nil. Kotlin의 null 안전성이 "값이 없음"을 타입으로 표시한다면, Go는 "항상 쓸 수 있는 기본값"을 줍니다.

    상수와 iota. const는 컴파일 시점 상수입니다. iota는 상수 묶음 안에서 0부터 1씩 늘어나는 값으로, 열거형(enum) 비슷한 값을 만들 때 씁니다.

    타입과 변환. 숫자 타입 사이에는 암묵적 변환이 없습니다. int와 float64를 섞으면 컴파일 오류이고, float64(count)처럼 직접 변환합니다. 문자열은 UTF-8 바이트의 나열이라 len("한글")은 글자 수가 아니라 바이트 수(6)를 돌려줍니다. 글자 단위는 []rune(문자열)로 바꿔 셉니다(rune은 유니코드 한 글자).

    제어문. if 조건에는 괄호를 쓰지 않고 중괄호는 필수입니다. if n := 계산(); n > 1 {처럼 조건 앞에 초기화문을 둘 수 있고, 그 변수는 if 블록 안에서만 보입니다. 반복문은 for 하나뿐이며 세 가지 모양(for 초기화; 조건; 증감, for 조건, for i := range 3)으로 for와 while을 모두 대신합니다. switch는 case가 끝나면 자동으로 빠져나옵니다(break 불필요). 한 case에 값을 여러 개 쓸 수 있고, 조건 없는 switch { case ...: }는 if-else 사슬을 읽기 좋게 씁니다.

    함수. 값을 여러 개 돌려줄 수 있고((int, error)), 반환값이 필요 없으면 _로 버립니다. 가변 인자는 nums ...int로 받고 슬라이스를 넘길 때는 values...로 펼칩니다. 함수는 값이라서 변수에 담거나 돌려줄 수 있고(클로저, closure), 바깥 변수를 기억합니다. 예외(exception) 대신 error 값을 돌려주는 방식은 레슨 05에서 자세히 다룹니다.

    5. 함수 프로그램 만들기

      lesson03/mine/functions 폴더와 main.go를 만들어 입력합니다. 함수 이름 divide를 입력하다가 자동 완성이 같은 파일 안의 함수를 보여 주는 것을 확인하세요.

      package main

import (
	"errors"
	"fmt"
)

func add(a, b int) int {
	return a + b
}

func divide(a, b int) (int, error) {
	if b == 0 {
		return 0, errors.New("0으로 나눌 수 없습니다")
	}
	return a / b, nil
}

func sum(nums ...int) int {
	total := 0
	for _, n := range nums {
		total += n
	}
	return total
}

func minMax(nums ...int) (lowest, highest int) {
	lowest, highest = nums[0], nums[0]
	for _, n := range nums {
		if n < lowest {
			lowest = n
		}
		if n > highest {
			highest = n
		}
	}
	return
}

func makeCounter() func() int {
	count := 0
	return func() int {
		count++
		return count
	}
}

func main() {
	fmt.Println("add(2, 3) =", add(2, 3))

	quotient, err := divide(7, 2)
	fmt.Println("7 / 2 =", quotient, err)

	_, err = divide(1, 0)
	fmt.Println("오류:", err)

	values := []int{4, 5, 6}
	fmt.Println("합계:", sum(), sum(1, 2, 3), sum(values...))

	lowest, highest := minMax(3, 9, 1)
	fmt.Println("최소/최대:", lowest, highest)

	next := makeCounter()
	fmt.Println("카운터:", next(), next(), next())
}

    함수 타입() -> IntCallable[[], int]func() int
    익숙함이 만드는 함정첫째, Kotlin의 if/when은 값을 돌려주는 표현식(val grade = if (score >= 80) "B" else "C")이지만, Go의 if와 switch는 문장이라 이렇게 쓸 수 없습니다. 변수를 먼저 선언하고 안에서 대입합니다. 둘째, val을 떠올리며 :=가 불변이라고 오해하기 쉽지만 :=는 단지 선언과 대입을 한 번에 하는 문법입니다. 셋째, zero value는 편하지만 "값이 없다"와 "0이다"가 구별되지 않습니다. 구별이 필요하면 다음 레슨의 포인터나 ok 패턴을 씁니다. 넷째, 블록 안에서 :=를 쓰면 바깥 변수가 아니라 새 변수가 만들어져, 코드가 컴파일되는데도 엉뚱한 변수를 바꾸는 실수가 생깁니다(100 Go Mistakes #1).
    val grade = if (score >= 80) "B" else "C"   // Kotlin: if 가 값을 돌려줍니다

    1. 인레이 힌트를 끄고 켜 보기

    문제 1. 평균 구하기

    숫자를 여러 개 받아 평균을 돌려주는 함수 average(nums ...float64) (float64, error)를 작성하세요. 값이 하나도 없으면 0과 오류를 돌려줍니다. 완료 조건: average(1, 2, 3, 4)는 2.5 <nil>을, average()는 오류 메시지를 출력한다.

    힌트 보기

    가변 인자 nums는 함수 안에서 슬라이스처럼 len(nums)와 for _, n := range nums로 다룹니다. 합계를 개수로 나눌 때 int인 len(nums)를 float64(len(nums))로 변환해야 합니다. 오류는 errors.New("메시지")로 만들고 import "errors"가 필요합니다.

    정답 예시 보기 
    func average(nums ...float64) (float64, error) {
	if len(nums) == 0 {
		return 0, errors.New("평균을 구할 값이 없습니다")
	}
	total := 0.0
	for _, n := range nums {
		total += n
	}
	return total / float64(len(nums)), nil
}

    문제 2. FizzBuzz

    1부터 15까지 숫자를 한 줄에 하나씩 출력하되, 3의 배수는 Fizz, 5의 배수는 Buzz, 15의 배수는 FizzBuzz로 출력하세요. 완료 조건: 정수 하나를 받아 문자열을 돌려주는 함수 fizzBuzz(n int) string을 만들고 switch(조건 없는 형태)를 사용한다. 출력 15줄이 아래 예상 출력과 같다.

    힌트 보기

    15의 배수 검사를 가장 먼저 해야 합니다. 3의 배수와 5의 배수 검사가 먼저 오면 15가 Fizz로 끝납니다. 정수를 문자열로 바꿀 때는 strconv.Itoa(n)을 씁니다(import "strconv"). 나머지는 n%3 == 0으로 검사합니다.

    정답 예시 보기 
    func fizzBuzz(n int) string {
	switch {
	case n%15 == 0:
		return "FizzBuzz"
	case n%3 == 0:
		return "Fizz"
	case n%5 == 0:
		return "Buzz"
	default:
		return strconv.Itoa(n)
	}
}

    FIX자주 나는 오류

    첫 번째와 세 번째 오류 출력은 위 6단계에서 mine/variables/main.go를 고친 결과와 같습니다. 나머지 세 개(함수 밖 :=, 반환값 개수, else 위치)는 설명용으로 만든 짧은 예제 코드에서 실행한 출력이라 파일 경로의 폴더 이름과 줄·열 번호가 내 파일과 다를 수 있습니다. 메시지 문장을 중심으로 읽으세요.

    증상: 변수를 선언만 하고 쓰지 않았다

    선언한 변수나 import를 쓰지 않으면 컴파일 오류입니다. 변수를 지우거나, 일부러 쓰지 않을 값이라면 _로 받습니다. 오류 줄 번호는 파일:줄:열 순서입니다.

    # golab/lesson03/mine/variables
lesson03\mine\variables\main.go:39:2: declared and not used: unused

    증상: 함수 밖에서 :=를 썼다

    :=는 함수 안에서만 쓸 수 있습니다. 함수 밖(패키지 수준)에서는 var message = "안녕"처럼 var로 선언합니다.

    # golab/lesson03/mine/variables
lesson03\mine\variables\main.go:5:1: syntax error: non-declaration statement outside function body

    증상: 서로 다른 숫자 타입을 연산했다

    Go는 숫자 타입을 자동으로 변환하지 않습니다. float64(count) * ratio처럼 한쪽을 명시적으로 변환합니다.

    # golab/lesson03/mine/variables
lesson03\mine\variables\main.go:39:14: invalid operation: count * ratio (mismatched types int and float64)

    증상: 반환값 개수가 맞지 않는다

    divide가 값 두 개를 돌려주는데 변수 하나로 받았습니다. result, err := divide(4, 2)처럼 둘 다 받거나, 하나를 버리려면 result, _ := divide(4, 2)를 씁니다. 오류를 버리는 것은 정말 괜찮을 때만 하세요.

    # golab/lesson03/mine/functions
lesson03\mine\functions\main.go:16:12: assignment mismatch: 1 variable but divide returns 2 values

    증상: else를 다음 줄에 썼다

    Go는 줄 끝에 세미콜론을 자동으로 넣기 때문에 } 다음 줄의 else가 앞 문장과 이어지지 않습니다. } else {처럼 같은 줄에 씁니다.

    # golab/lesson03/mine/control
lesson03\mine\control\main.go:10:2: syntax error: unexpected keyword else, expected }
