    핵심 개념 한국어 요약

    계층(layer)과 책임. handler는 HTTP를 아는 유일한 계층으로 요청을 읽고(경로 변수, JSON), 서비스를 호출하고, 결과를 상태 코드와 JSON으로 바꿉니다. service는 업무 규칙(제목 공백 제거, 100자 제한, 수정할 필드가 없으면 거부)을 담고 HTTP도 저장 방식도 모릅니다. repository는 저장과 조회만 합니다. 이렇게 나누면 handler 테스트는 가짜 서비스로, service 테스트는 메모리 저장소로 서버 없이 빠르게 돌릴 수 있습니다(Learn Go with Tests의 "핸들러가 하는 일은 파싱·서비스 호출·응답 세 가지"라는 정리).

    의존성 주입(dependency injection). Spring처럼 컨테이너가 객체 그래프를 만들어 주지 않습니다. main에서 repository.NewMemory() → service.New(repo) → handler.New(svc)처럼 생성자 호출로 직접 연결합니다. 모든 부품을 조립하는 이 자리를 composition root라고 부릅니다. 의존성이 인터페이스 타입 필드이므로 테스트에서는 같은 자리에 가짜를 넣을 수 있습니다.

    인터페이스는 사용하는 쪽에서. Go의 인터페이스는 구현 선언 없이 메서드만 맞으면 충족됩니다. 그래서 인터페이스를 쓰는 쪽 패키지에 필요한 만큼만 정의합니다. 이 프로젝트에서 service.Repository는 service 패키지에, handler.TaskService는 handler 패키지에 있고, 구현체(repository.Memory, service.Service)는 그 인터페이스를 import하지도 않습니다. 구현은 구체 타입으로 돌려주고 받는 쪽이 인터페이스로 받는 것이 기본입니다(100 Go Mistakes #5~#7).

    미들웨어(middleware). http.Handler를 받아 http.Handler를 돌려주는 함수입니다. 안쪽 핸들러를 부르기 전후에 로깅, 패닉 복구, 인증 같은 공통 작업을 끼워 넣습니다. 여러 개를 겹치면 먼저 적은 것이 가장 바깥쪽입니다.

    httptest. httptest.NewRequest로 가짜 요청을 만들고 httptest.NewRecorder에 응답을 받아 상태 코드·헤더·본문을 검사합니다. 네트워크도 포트도 쓰지 않아 빠르고, handler.ServeHTTP(rec, req)를 직접 부릅니다. 진짜 서버가 필요한 흐름 테스트는 httptest.NewServer로 임시 서버를 띄웁니다.

    레이어 이름 패키지에 대하여. Go에서는 패키지를 기능(context)별로 나누는 것을 선호하는 사람도 많고, 계층(layer)별로 나누는 것도 일관되기만 하면 틀린 선택이 아닙니다(100 Go Mistakes #12). 이 코스는 Spring 경험을 살려 계층 이름을 그대로 썼습니다.

      func TestCreateValidation(t *testing.T) {
	tests := []struct {
		name      string
		title     string
		wantTitle string
		wantErr   bool
	}{
		{"정상", "Go 공부하기", "Go 공부하기", false},
		{"앞뒤 공백 제거", "  장보기  ", "장보기", false},
		{"빈 문자열", "", "", true},
		{"공백만", "   ", "", true},
		{"한글 100자는 허용", strings.Repeat("가", 100), strings.Repeat("가", 100), false},
		{"한글 101자는 거부", strings.Repeat("가", 101), "", true},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got, err := newTestService().Create(context.Background(), tt.title)

			var validation *domain.ValidationError
			if tt.wantErr {
				if !errors.As(err, &validation) {
					t.Fatalf("ValidationError가 와야 합니다: %v", err)
				}
				return
			}
			if err != nil {
				t.Fatalf("예상하지 못한 에러: %v", err)
			}
			if got.Title != tt.wantTitle {
				t.Errorf("Title = %q, want %q", got.Title, tt.wantTitle)
			}
		})
	}
}

    6. 미들웨어 읽고 테스트하기, 쌓는 순서 확인하기

      mine/internal/handler/middleware.go를 엽니다. Chain은 목록의 첫 번째가 가장 바깥쪽이 되도록 뒤에서부터 감쌉니다. Logging은 안쪽 핸들러가 끝난 뒤 상태 코드와 걸린 시간을 기록하려고 ResponseWriter를 감싼 statusRecorder를 씁니다. Recover는 defer 안에서 recover()로 패닉을 잡아 500 응답으로 바꿉니다.

      // Middleware는 핸들러를 감싸 공통 기능을 덧붙이는 함수입니다.
type Middleware func(http.Handler) http.Handler

// Chain은 h를 mws로 감쌉니다. 목록의 첫 번째가 가장 바깥쪽입니다.
func Chain(h http.Handler, mws ...Middleware) http.Handler {
	for i := len(mws) - 1; i >= 0; i-- {
		h = mws[i](h)
	}
	return h
}
      // Recover는 핸들러에서 난 패닉을 잡아 500 응답으로 바꿉니다.
func Recover(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		defer func() {
			if v := recover(); v != nil {
				log.Printf("패닉 복구: %v", v)
				writeError(w, http.StatusInternalServerError, "서버 내부 오류")
			}
		}()
		next.ServeHTTP(w, r)
	})
}

    7. 소스 제어 보기 읽기 (diff, stage, commit의 흐름)

      Ctrl+Shift+G를 눌러 소스 제어(Source Control) 보기를 엽니다. 이 단계는 읽기만 합니다. 이 코스 폴더 자체는 Git 저장소가 아닐 수 있고(이 PC에서는 코스 폴더에 .git이 없습니다), 그 경우 저장소를 새로 만들자는 안내가 보일 수 있는데 지금은 누르지 않아도 됩니다. 클론한 자료 폴더 안에는 각자의 .git이 있으니 그 저장소들이 목록에 나타날 수도 있습니다. 읽기 전용 자료이므로 어떤 파일도 스테이지하거나 커밋하지 마세요. 자신의 다른 프로젝트 폴더에서 같은 화면을 열어 보면 아래 흐름이 그대로 보입니다(VS Code 공식 문서의 "Staging and committing" 기준).

      ① 변경(Changes) 목록에 저장한 파일이 나타납니다. 파일 이름 옆 글자는 M(수정), U(새 파일, 추적 전), D(삭제)입니다. ② 파일을 클릭하면 diff 편집기가 열려 원래 내용과 바뀐 내용을 나란히 보여 줍니다. 레슨 08의 마지막 단계에서 쓴 "비교를 위해 선택/선택 항목과 비교"와 같은 모양입니다. ③ 파일에 마우스를 올려 +를 누르면 스테이징(stage)되어 스테이지된 변경(Staged Changes)으로 옮겨 갑니다. 스테이징은 "다음 커밋에 넣을 내용을 고르는" 단계이고, 그 뒤에 고친 내용은 다시 스테이징하기 전까지 커밋에 들어가지 않습니다. ④ 위쪽 메시지 칸에 한 줄 요약을 쓰고 커밋(Commit)을 누르면 로컬 기록에 남습니다. 서버로 올리는 push는 별개의 동작입니다. ⑤ 실수로 지우고 싶지 않은 변경은 변경 취소(Discard Changes)를 누르기 전에 반드시 확인하세요. 저장되지 않은 파일이 사라질 수 있습니다.

      지금 하고 있는 "복사본에서 먼저 고치고, 원본과 비교한다"는 흐름이 Git 브랜치와 diff가 자동화해 주는 일입니다. Spring 프로젝트에서 쓰던 IntelliJ의 변경 목록(Changes) 화면과 같은 개념이라고 보면 됩니다.

    문제 2. API 키를 검사하는 미들웨어 만들기

    mine/exercises 폴더에 apikey.go를 만들어 RequireAPIKey(key string) func(http.Handler) http.Handler를 구현하세요. X-API-Key 헤더가 key와 같은 요청만 통과시키고, 아니면 401과 {"error":"인증이 필요합니다"}를 돌려주며, GET /health는 키 없이도 통과시킵니다. 완료 조건: 아래 테스트를 apikey_test.go로 입력해 go test ./lesson09/mine/exercises가 통과한다.

    package exercises

import (
	"net/http"
	"net/http/httptest"
	"testing"

func TestRequireAPIKey(t *testing.T) {
	ok := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
	})
	protected := RequireAPIKey("secret")(ok)

	tests := []struct {

		method     string
		path       string

		{"키가 틀리면 401", http.MethodPost, "/tasks", "wrong", http.StatusUnauthorized},
		{"health는 키 없이 통과", http.MethodGet, "/health", "", http.StatusOK},

			}
		})
	}
}
    힌트 보기

    미들웨어는 "설정 값을 받아 미들웨어를 돌려주는" 함수 한 겹이 더 있는 모양입니다. RequireAPIKey("secret")이 func(http.Handler) http.Handler를 돌려주고, 그것이 다시 http.HandlerFunc를 돌려줍니다. 통과시키려면 next.ServeHTTP(w, r)를 부르고, 막으려면 응답을 쓴 뒤 반드시 return합니다. 비밀 값 비교는 시간차 공격을 줄이려고 crypto/subtle의 ConstantTimeCompare를 쓰는 것이 관례입니다.

    정답 예시 보기 
    // Package exercises는 레슨 09 연습 문제의 정답 예시입니다.
package exercises

import (
	"crypto/subtle"
	"net/http"
)

// RequireAPIKey는 X-API-Key 헤더가 key와 같은 요청만 통과시킵니다.
// GET /health는 상태 확인용이므로 키 없이도 통과시킵니다.
func RequireAPIKey(key string) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if r.Method == http.MethodGet && r.URL.Path == "/health" {
				next.ServeHTTP(w, r)
				return
			}
			got := r.Header.Get("X-API-Key")
			if subtle.ConstantTimeCompare([]byte(got), []byte(key)) != 1 {
				w.Header().Set("Content-Type", "application/json")
				w.WriteHeader(http.StatusUnauthorized)
				_, _ = w.Write([]byte(`{"error":"인증이 필요합니다"}` + "\n"))
				return
			}
			next.ServeHTTP(w, r)
		})
	}
}

    3단계에서 import 경로를 바꾸지 않아 복사본의 main이 원본 internal을 쓰고 있는 경우입니다. 오류가 나지 않아서 더 헷갈립니다. 전체 검색(Ctrl+Shift+F)에서 golab/lesson09/internal을 포함할 파일 ./lab/lesson09/mine로 검색해 결과가 하나도 없어야 정상입니다. 남아 있으면 3단계의 전체 바꾸기를 다시 합니다.

    증상: 전체 바꾸기를 눌렀더니 다른 레슨까지 바뀌었다

    포함 칸을 비우고 모두 바꾸기를 누르면 열어 둔 폴더 전체가 대상이 됩니다. 이런 일이 생기지 않게 하려면 바꾸기 전에 ① 포함 칸에 폴더를 적고 ② 결과의 파일 수가 예상(7개)과 맞는지 보고 ③ 결과 줄을 클릭해 diff를 확인합니다. 이 코스에서는 lab/lesson09의 완성 코드와 레슨 09 페이지(lessons/09-taskapi-layers.html)의 코드 블록까지 바뀝니다(레슨 10, 11은 경로의 숫자가 달라 영향이 없습니다). 저장 전이라면 바뀐 파일마다 편집기에서 Ctrl+Z로 되돌릴 수 있습니다. 이미 저장했다면 같은 검색 보기에서 찾을 칸과 바꿀 칸을 서로 맞바꿔 한 번 더 바꾸면 되돌아가는 경우가 많지만 확실하지 않으니, 처음부터 포함 칸을 채우는 습관이 최선입니다.
