    핵심 개념 한국어 요약

    계약 테스트(contract test). 같은 인터페이스를 만족하는 구현이 둘 이상일 때, "이 인터페이스를 구현한 것이라면 반드시 지켜야 하는 약속"을 테스트 함수 하나(runContract)로 쓰고 구현마다 그 함수를 호출합니다. 메모리 저장소와 파일 저장소가 같은 약속(만든 것을 다시 읽을 수 있다, 없는 것은 ErrNotFound, 목록은 ID 순서이고 쪽 나누기를 지킨다)을 지키는지 한 번에 확인하고, 나중에 데이터베이스 저장소를 추가해도 같은 함수에 한 줄만 더하면 됩니다.

    쪽 나누기(pagination)와 전체 개수. limit은 한 쪽에 돌려줄 최대 개수, offset은 앞에서 건너뛸 개수입니다. 쪽을 자르기 전의 전체 개수(조건에 맞는 모든 항목 수)는 응답 본문 대신 X-Total-Count 헤더로 알려 주면 본문은 계속 배열 그대로라 기존 클라이언트가 깨지지 않습니다. 범위 검사(limit 1~100, offset 0 이상)는 업무 규칙이라 service에 두고, 문자열 숫자 변환 같은 입력 형식 검사는 handler에 둡니다.

    인터페이스를 바꾸면 컴파일러가 안내합니다. service.Repository의 List 시그니처를 바꾸면 그 인터페이스를 쓰거나 구현하는 모든 곳이 컴파일 오류가 됩니다. 오류가 곧 "고칠 곳 목록"이라서, 테스트 → 시그니처 → 구현 → 호출하는 쪽 순서로 오류가 줄어드는 것을 따라가면 됩니다. 인터페이스를 쓰는 쪽에서 정의한 덕분에 영향 범위도 service와 그 구현체, handler로 한정됩니다.

      // ListFilter는 목록 조회 조건입니다.
type ListFilter struct {
	Done   *bool  // nil이면 완료 여부와 상관없이 모두 조회합니다.
	Query  string // 비어 있지 않으면 제목에 이 문자열이 들어 있는(대소문자 무시) Task만 조회합니다.
	Limit  int    // 한 번에 돌려줄 최대 개수입니다. 0이면 서비스의 기본값을 씁니다.
	Offset int    // 앞에서부터 건너뛸 개수입니다.
}

// Page는 목록 조회 결과 한 쪽입니다.
type Page struct {
	Tasks []Task

func parseListFilter(q url.Values) (domain.ListFilter, error) {
	filter := domain.ListFilter{Query: q.Get("q")}

	switch v := q.Get("done"); v {
	case "":
	case "true", "false":

		}

	}
	return filter, nil

    package repository

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"sync"

	"golab/lesson11/internal/domain"
)

// File은 Memory 위에 JSON 파일 저장을 얹은 저장소입니다.
// 서버를 다시 시작해도 Task가 남고, 변경할 때마다 파일 전체를 다시 씁니다.
type File struct {
	mu   sync.Mutex
	path string
	mem  *Memory
}

type fileData struct {
	NextID int           `json:"next_id"`
	Tasks  []domain.Task `json:"tasks"`
}

// NewFile은 path의 데이터를 읽어 저장소를 만듭니다. 파일이 없으면 빈 상태로 시작합니다.
// 시작할 때 한 번 저장해 보므로, 폴더가 없거나 쓸 수 없으면 서버가 뜨기 전에 에러가 납니다.
func NewFile(path string) (*File, error) {
	f := &File{path: path, mem: NewMemory()}
	if err := f.load(); err != nil {
		return nil, err
	}
	if err := f.save(); err != nil {
		return nil, err
	}
	return f, nil
}

func (f *File) load() error {
	b, err := os.ReadFile(f.path)
	if errors.Is(err, os.ErrNotExist) {
		return nil
	}
	if err != nil {
		return fmt.Errorf("데이터 파일을 읽을 수 없습니다: %w", err)
	}

	var data fileData
	if err := json.Unmarshal(b, &data); err != nil {
		return fmt.Errorf("데이터 파일 형식이 올바르지 않습니다: %w", err)
	}
	f.mem.restore(data.NextID, data.Tasks)
	return nil
}

// save는 임시 파일에 쓴 뒤 이름을 바꿔, 쓰다가 멈춰도 기존 파일이 깨지지 않게 합니다.
func (f *File) save() error {
	nextID, tasks := f.mem.snapshot()
	b, err := json.MarshalIndent(fileData{NextID: nextID, Tasks: tasks}, "", "  ")
	if err != nil {
		return err
	}

	tmp := f.path + ".tmp"
	if err := os.WriteFile(tmp, b, 0o644); err != nil {
		return fmt.Errorf("데이터 파일을 쓸 수 없습니다: %w", err)
	}
	if err := os.Rename(tmp, f.path); err != nil {
		return fmt.Errorf("데이터 파일을 교체할 수 없습니다: %w", err)
	}
	return nil
}

func (f *File) List(ctx context.Context, filter domain.ListFilter) (domain.Page, error) {
	return f.mem.List(ctx, filter)
}

func (f *File) Get(ctx context.Context, id int) (domain.Task, error) {
	return f.mem.Get(ctx, id)
}

func (f *File) Create(ctx context.Context, title string) (domain.Task, error) {
	f.mu.Lock()
	defer f.mu.Unlock()

	t, err := f.mem.Create(ctx, title)
	if err != nil {
		return domain.Task{}, err
	}
	return t, f.save()
}

func (f *File) Update(ctx context.Context, id int, in domain.UpdateInput) (domain.Task, error) {
	f.mu.Lock()
	defer f.mu.Unlock()

	t, err := f.mem.Update(ctx, id, in)
	if err != nil {
		return domain.Task{}, err
	}
	return t, f.save()
}

func (f *File) Delete(ctx context.Context, id int) error {
	f.mu.Lock()
	defer f.mu.Unlock()

	if err := f.mem.Delete(ctx, id); err != nil {
		return err
	}
	return f.save()
}

func TestParseListFilter(t *testing.T) {
	yes, no := true, false
	tests := []struct {
		name    string
		query   string
		want    domain.ListFilter
		wantErr bool
	}{
		{"아무것도 없음", "", domain.ListFilter{}, false},
		{"done=true", "done=true", domain.ListFilter{Done: &yes}, false},
		{"done=false", "done=false", domain.ListFilter{Done: &no}, false},
		{"모두 지정", "done=true&q=go&limit=10&offset=20", domain.ListFilter{Done: &yes, Query: "go", Limit: 10, Offset: 20}, false},
		{"done이 true/false가 아님", "done=maybe", domain.ListFilter{}, true},
		{"limit이 숫자가 아님", "limit=abc", domain.ListFilter{}, true},
		{"offset이 숫자가 아님", "offset=1.5", domain.ListFilter{}, true},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			values, err := url.ParseQuery(tt.query)
			if err != nil {
				t.Fatal(err)
			}
			got, err := parseListFilter(values)
			if (err != nil) != tt.wantErr {
				t.Fatalf("err = %v, wantErr = %v", err, tt.wantErr)
			}
			if !tt.wantErr && !reflect.DeepEqual(got, tt.want) {
				t.Errorf("got %+v, want %+v", got, tt.want)
			}
		})
	}
}
