use std::sync::mpsc;
use std::thread;

fn main() {
    let (tx, rx) = mpsc::channel();
    for worker in 0..3 {
        let tx = tx.clone();
        thread::spawn(move || {
            tx.send(worker * 100).unwrap();
        });
    }

    for value in rx.iter() {
        println!("받음: {value}");
    }
    println!("모두 받았습니다");
}
